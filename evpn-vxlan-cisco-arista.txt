==========================Spin-10==========================
conf t
hostname Spin-10
ip routing
ip vrf management

vlan 99
name RED-OOB-GESTION-1

vlan 88
name RED-OOB-GESTION-2

spanning-tree mode rapid-pvst
spanning-tree vlan 99 priority 4096
spanning-tree vlan 88 priority 8192

int Vlan99
ip vrf forwarding management
ip address 172.16.99.10 255.255.255.0
no shu

int Vlan88
ip vrf forwarding management
ip add 172.16.88.10 255.255.255.0
no shu

int range gi0/0 - 3
switchport tr encapsulation dot1q
switchport mode tr
no shu

ip sla 1
icmp-echo 172.16.99.1 source-interface Vlan99
vrf management
frequency 5
ip sla schedule 1 life forever start-time now
track 1 ip sla 1 reachability

ip route vrf management 0.0.0.0 0.0.0.0 172.16.99.1 track 1
ip route vrf management 0.0.0.0 0.0.0.0 172.16.88.1 5
ip ssh version 2

end
wr
